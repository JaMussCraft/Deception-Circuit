 def test_transport_preserves_slurm_and_writes_only_private_child(self):
  cfg=config(Path('/projects/beto/swong2/Alignment/run311'))
  argv=h.container_command(cfg,'v1',{'SLURM_JOB_ID':'123','SLURM_CPUS_PER_TASK':'128'},9999)

  self.assertIn('A3_WRAPPER_START_SECONDS=9999',argv)
