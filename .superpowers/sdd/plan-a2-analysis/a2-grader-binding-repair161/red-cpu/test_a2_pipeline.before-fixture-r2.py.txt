def rehash(value):
    value["comparison_sha256"] = comparison.digest(
        {k: v for k, v in value.items() if k != "comparison_sha256"}
    )
    return value

@pytest.mark.parametrize(
    "change", ["missing_source", "stale_hash", "wrong_X", "changed_id", "stale_sidecar"]
)
def test_freeze_fails_closed(inputs, change):
    comp, p, s, sidecar = inputs
    if change == "missing_source":
        comp["sources"].pop()
        rehash(comp)
    elif change == "stale_hash":
        comp["sources"][0]["seed"] = 999
    elif change == "wrong_X":
        comp["tasks"]["far_v2"]["seeds"]["42"]["X"] = comparison._group(set())
        rehash(comp)
    elif change == "changed_id":
        p["designs"]["far_v2"]["assessment_ids"] = ["changed"]
    else:
        sidecar["selection_file_sha256"] = "bad"
    with pytest.raises(ValueError):
        freeze(inputs)

            a2_receipt_sha256=receipt["receipt_sha256"],

def test_task_aggregate_requires_all_seeds_floors_and_stability(inputs):
    receipt = freeze(inputs)
    cells = [
        dict(
            task=g["task"],
            group=name,
            operation="item_swap",
            point_pass=True,
            strong=True,
            complete_assay_ready=True,
            assay_readiness={"MMLU": {"passed": True}},
        )
        for name, g in receipt["groups"].items()
    ]
    out = a2.aggregate_tasks(receipt, {"cells": cells})
    assert all(t["robust_task_pass"] for t in out["tasks"].values())
    assert all(t["strong"] for t in out["tasks"].values())
    for field in ("point_pass", "strong", "complete_assay_ready", "floor"):
        changed = copy.deepcopy(cells)
        if field == "floor":
            changed[0]["assay_readiness"]["MMLU"]["passed"] = False
        else:
            changed[0][field] = False
        task = a2.aggregate_tasks(receipt, {"cells": changed})["tasks"]["far_v2"]
        assert task["robust_task_pass"] == (field == "strong")
        assert not task["strong"]
    assert not a2.aggregate_tasks(receipt, {"cells": cells[1:]})["tasks"]["far_v2"][
        "robust_task_pass"
    ]
    bad = copy.deepcopy(receipt)
    bad["comparison"]["tasks"]["far_v2"]["pairwise"]["mixed"][2]["ratio"] = 0.49
    assert not a2.aggregate_tasks(bad, {"cells": cells})["tasks"]["far_v2"][
        "robust_task_pass"
    ]

def test_tiny_actual_hooks_new_bank_scoring_and_references(inputs, tmp_path):
    import torch
    import nnsight

    if nnsight.__version__ != "0.7.0":
        pytest.skip("pinned NNsight0.7.0 required")
    from nnsight import NNsight
    from transformers import LlamaForCausalLM
    from tests.helpers import tiny_config

    torch.manual_seed(41)
    model = NNsight(LlamaForCausalLM(tiny_config(attn_implementation="sdpa")).eval())
    receipt = freeze(inputs)
    prepared = inputs[1]
    args = SimpleNamespace(batch_size=2, limit=1, injected=True, commit="b" * 40)
    banks = a2.collect_banks(args, receipt, prepared, model)
    path = tmp_path / "banks.json"
    a2.publish_artifact(path, banks)
    a2.validate_banks(receipt, prepared, banks)
    assert all(b["provenance"]["n_tokens"] == 16 for b in banks["banks"].values())
    result = a2.score(args, receipt, prepared, banks, path, model)
    assert result["counts"]["rows"] == len(a2.expected_keys(receipt, 1))
    assert result["counts"]["positive_controls"] == 36
    assert len({r["group"] for r in result["rows"]}) == 9
    a2.validate_a2_scope(receipt, result["rows"], limit=1, banks=banks, bank_path=path)
    refs = [r for r in result["rows"] if r["operation"] == "h_mean"]
    assert refs and all(
        r["mean_bank_provenance"]["owner"] == "a2_original_H_union" for r in refs
    )
    assert result["execution_provenance"]["device"] == "cpu"
    refs[0]["mean_bank_provenance"]["bank_file_sha256"] = "bad"
    with pytest.raises(ValueError, match="bank row"):
        a2.validate_a2_scope(
            receipt, result["rows"], limit=1, banks=banks, bank_path=path
        )


def test_summary_keeps_floor_failure_separate_from_primary_rule(inputs):
    receipt = freeze(inputs)
    rows = synthetic_rows(receipt)
    # A primary operation can pass while the joined H-mean MMLU floor fails.
    for r in rows:
        if (
            r["condition"] == "MMLU"
            and r["operation"] == "h_mean"
            and r["instance_id"] in ("mmlu0", "mmlu1")
        ):
            r["patched_correct"] = False
    controls = [
        dict(
            zip(("mechanism_task", "task", "instance_id", "condition", "stream"), key),
            valid=True,
        )
        for key in a1.positive_keys(a2._prepared_domain(receipt))
    ]
    scored = dict(
        rows=rows,
        positive_controls=controls,
        limit=0,
        pilot=False,
        injected=True,
        receipt_sha256=receipt["receipt_sha256"],
    )
    out = a2.summarize(receipt, scored, n_boot=8)
    task = out["tasks"]["far_v2"]
    seed = task["per_seed"]["42"]
    assert seed["primary_point_pass"]
    assert seed["complete_assay_ready"]
    assert not seed["both_mmlu_floors_pass"]
    assert not task["robust_task_pass"]
    assert (
        out["summary"]["bootstrap"]["unit"]
        == "source pressure id; paired id for utility"
    )
    with pytest.raises(ValueError):
        a2.summarize(receipt, dict(scored, positive_controls=controls[:-1]), n_boot=8)

def _catalogue_edit(path, edit):
    data = json.loads(path.read_text())
    edit(data)
    path.write_text(json.dumps(data))
